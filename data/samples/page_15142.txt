    function mytrim(s){
        return ( s || '' ).replace( /^\s+|\s+$/g, '' ).replace(/\\"/g,'"');
    }

    function build_page_individuallisting(mydata) {
        var listing_name=(mydata.listing_name).replace("\\","");
        var contributor_link=contributor_link_head+"?lid="+mydata.listing_id+"&dt="+mydata.directory_type;
        var report_link=report_link_head+"?lid="+mydata.listing_id+"&dt="+mydata.directory_type;
        var claimlisting_link=claimlisting_link_head+"?lid="+mydata.listing_id+"&dt="+mydata.directory_type;
        var update_time=mydata.update_time;
        var mycolor=get_directory_color(mydata.directory_type);
        var mydtype_name=get_directory_fullname(mydata.directory_type);
        var directory_tag= "<span style='font-size:0.95rem;padding:5px; color:#fff; background-color: " +mycolor+ "'>"+mydtype_name+"</span>";
        if (mydata.image) {
            //var myimage=image_path+"/"+mydata.image;
            var myimage = "<img alt='Listing Image' src='" +image_path+"/"+mydata.image+ "'></img>";
        } else {
            //var myimage=image_path+"/default-"+mydata.directory_type+"-4-3.jpg";
            var myimage = "<img  alt='Listing Image' src='" +image_path+"/default-"+mydata.directory_type+"-4-3.jpg"+ "'></img>";
        }
        var num_lines=0;
        var left_ct_1=[];
        var right_ct_1=[];
        var tmp_sep="";
        if (mydata.soical_media) {
            left_ct_1.push(mydata.soical_media);
            num_lines=num_lines+1;
        }
        var webscripting_html="";
        if (mydata.webscriping_status=="yes") {
            webscripting_html=`<span class="claimlisting">
                               <a href="${claimlisting_link}" class="button is-rounded is-ghost  get-direction-link">
                                   <i class="fas fa-share-alt-square"></i>&nbsp;Claim This Listing
                               </a>
                           </span>`;
        }
        //console.log(mydata.listing_desc);
        if (mydata.listing_desc) {
            left_ct_1.push("<div class='listing_truncate  mytext margin_afterdiv'>"  +mytrim(mydata.listing_desc)+ "</div>");
            //num_lines=num_lines+Math.floor(mydata.listing_desc.length/45,0);
        }
        if (mydata.address) {
            mydata.address=mytrim(mydata.address);
            if (num_lines<13) {
                num_lines=num_lines+(mydata.address.split("<div>")).length - 1;
                var tmp12=mydata.address.split("<div class='locationdesc");
                if (tmp12.length==2) {
                    var tmp13=tmp12[1].split("</div>");
                    num_lines=num_lines+Math.floor(tmp13[1].length/45,0)-1;
                }
                left_ct_1.push(mydata.address);
            } else {
                right_ct_1.push(mydata.seasonproducts);
            }
        }

        if (mydata.contact_name) {
            if (num_lines<13) {
                left_ct_1.push(tmp_sep+mydata.contact_name);
            } else {
                right_ct_1.push(tmp_sep+mydata.contact_name);
            }
            num_lines=num_lines+2;
        }
        if (mydata.contact_email) {
            if (num_lines<13) {
                left_ct_1.push(tmp_sep+mydata.contact_email);
            } else {
                right_ct_1.push(tmp_sep+mydata.contact_email);
            }
            num_lines=num_lines+1;
        }
        if (mydata.contact_phone) {
            if (num_lines<13) {
                left_ct_1.push(tmp_sep+mydata.contact_phone);
            } else {
                right_ct_1.push(tmp_sep+mydata.contact_phone);
            }
            num_lines=num_lines+1;
        }

        if (mydata.seasonproducts) {
            right_ct_1.push(tmp_sep+mydata.seasonproducts);
        }
        if (mydata.specialproductionmethods) {
            right_ct_1.push(tmp_sep+mydata.specialproductionmethods);
        }

        if (mydata.acceptedpayment) {
            right_ct_1.push(mydata.acceptedpayment);
        }
        if (mydata.fnap) {
            right_ct_1.push(mydata.fnap);
        }
        if (mydata.saleschannel) {
            right_ct_1.push(mydata.saleschannel);
        }
        if (mydata.profile) {
            right_ct_1.push(mydata.profile);
        }
        if (mydata.customizedcontnet) {
            right_ct_1.push(mydata.customizedcontnet);
        }

        let left_ct=left_ct_1.join(" ");
        let right_ct=right_ct_1.join(" ");


        return `
        <div class="container mypadding" style=" margin-left: auto;margin-right: auto;">
            <div class="flistingcontainer card">
                <div class="listing_name">
                   <div class="container">
                           <div class="title is-size-4-desktop is-size-5-tablet is-size-5-mobile">
                                 ${listing_name}
                           </div>
                   </div>
                </div> ${ /* 1*/''}
                <div class="directorytype">${directory_tag}</div> ${ /* 2*/''}

                <div class="myhr">
                    <hr style="padding: 0 0 0px 0; margin: 0 0 10px 0;">
                </div> ${ /* 4*/''}
                <div class="basicinfo  is-size-4-desktop is-size-5-tablet is-size-5-mobile">
                   ${left_ct}
                </div> ${ /* 5*/''}
                <div class="image">${myimage}</div>${ /* 6*/''}
                <div class="otherinfo  is-size-4-desktop is-size-5-tablet is-size-5-mobile">
                   ${right_ct}
                </div>
                <div class="updatedate">
                    <hr style="padding: 0 0 0px 0; margin: 0 0 10px 0;">
                     <div class="statuscontainer ">
                            ${webscripting_html}
                            <span class="contributor">
                               <a href="${contributor_link}" class="button is-rounded is-ghost  get-direction-link">
                                   <i class="fas fa-share-alt-square"></i>&nbsp;Be a contributor
                               </a>
                           </span>
                           <span class="report">
                               <a href="${report_link}"  class="button is-rounded is-ghost  get-direction-link">
                                   <i class="fas fa-exclamation-circle"></i>&nbsp;Report&nbsp;Problems
                               </a>
                           </span>
                           <div style="padding-right: 20px; padding-bottom: 5px;">Latest Update: <span>${update_time}</span></div>
                     </div>
                </div>${ /* 3*/''}
            </div>
        </div>
        `;
    }
